## 5. 남은 문제

1. **대표님 PC에서 프로그램이 예전처럼 켜지는지 확인 필요**(새 폴더에 풀어 실행, 실제 저장 없이)
2. `job-log`·`number-check`는 **등록 흐름에 아직 연결 안 함** — 화면에서 번호·목록을 읽는 자리를 10/19 화면 사진으로 정한 뒤 연결
3. 자동 로그인이 다른 host로 넘어가는 로그인 방식이면 이제 멈춤(안전 쪽). 실제로 그렇다면 그 host를 허용 목록에 넣어야 함
4. 사이트 팝업은 http·https 를 모두 허용(우편번호 찾기 같은 바깥 화면이 깨질까 봐). 실제 쓰는 팝업 주소를 모은 뒤 더 좁힐 예정
5. 사진 경고창은 화면이 새로 그려지면 모아 둔 내용이 사라질 수 있음(다음에 개발자 연결로 경고창을 직접 받는 방식 검토)
6. 다음 차례: 창 튀어나옴(백그라운드 시험) · 엔트리 연결 · 화면 시안 · 번호판 사진 읽기

## 6. 검증해 주셨으면 하는 점

1. **두 번 등록 막기**: «저장 단추를 풀기 전에 기록을 디스크에 남기고(fsync), 남기지 못하면 풀지 않는다 · 다시 켰을 때 submitReady 에서 멈춘 전산은 확인 전까지 다시 저장 불가» 설계에 빈틈이 있는지
2. **한 번에 한 대 잠금**: «저장을 시작한 작업이 닫히기 전에는 다음 차 저장 불가»가 충분한지(작업을 닫는 조건 포함)
3. **예전 점검 구분**: 저장 전 찍어 둔 번호 목록 + 지운 줄 제외 + 작업 시작일 이후 + 차량번호·차대번호 일치 + 정확히 1개 규칙에 빠진 경우가 있는지
4. **차대번호 일부 비교**: 목록에 차대번호 일부만 보일 때 «뒤 6자리 이상 같으면 같음, 5자리 이하는 모름» 기준이 적당한지
5. **보안 수정**(비밀번호 넣을 주소, 팝업·이동·권한 제한, 기록 칸·기간)이 충분한지
6. 0장 고정 결정은 **검증 범위 밖**입니다(대표님 결정).

 *  · 기록 파일에 남겨도 되는 칸만 남기기 + 오래된 기록 지우기
 *
 *  나눔 원칙(상용화 대비): 이 파일은 «업무 관리» 칸에 속한다. 특정 전산 이름을 모른다.
 */

/** 주소에서 host 를 꺼낸다. 못 꺼내면 '' */
function hostOf(url) {
  try { return new URL(String(url || '')).host.toLowerCase(); } catch { return ''; }
}

/**
 * 비밀번호를 넣어도 되는 주소인가.
 *  https 이고, host 가 그 사이트 홈 주소의 host 와 정확히 같을 때만.
 *  (비슷한 이름의 다른 주소 — 예: ai.carinfo.co.kr.evil.com — 는 안 됨)
 */
function isSiteUrl(url, home) {
  let u, h;
  try { u = new URL(String(url || '')); h = new URL(String(home || '')); } catch { return false; }
  if (u.protocol !== 'https:') return false;
  return u.host.toLowerCase() === h.host.toLowerCase();
}

/**
 * 사이트 창이 새 창(팝업)으로 열어도 되는 주소인가.
 *  빈 창(about:blank)은 허용 — 카인포 사진등록처럼 빈 창을 먼저 띄우고 form 을 보내는 화면이 있다.
 *  http·https 는 허용 (사이트가 쓰는 우편번호 찾기 같은 바깥 화면이 있을 수 있어 막지 않는다).
 *  그 밖의 모양(javascript:·file:·data: 등)은 막는다.
 */
function popupAllowed(url) {
  const s = String(url || '').trim();
  if (s === '' || /^about:blank$/i.test(s)) return true;
  return /^https?:\/\//i.test(s);
}

/** 기록 파일에 남기지 않는 칸 — 개인정보와 PC 안 파일 위치 */
const DROP_KEYS = ['clientName', 'dealerHp'];
const DROP_PATTERNS = [/^photo\d+$/];

/** 기록에 남길 입력값만 고른다 (원래 값은 건드리지 않는다) */
function sanitizeRecordData(data) {
  const out = {};
  Object.keys(data || {}).forEach((k) => {
    if (DROP_KEYS.includes(k)) return;
    if (DROP_PATTERNS.some((re) => re.test(k))) return;
    out[k] = data[k];
  });
  return out;
}

/**
 * 기록 줄 가운데 보관 기간(일)이 지난 것을 뺀다.
 *  날짜를 못 읽는 줄은 남긴다(함부로 지우지 않음).
 */
function pruneRecordLines(lines, days, now = Date.now()) {

    try {

    }

}

    },

    /**
     * [저장] 단추를 풀어도 되는가.
     *  - 이미 저장을 풀어 준 적이 있으면(불확실 포함) 안 됨 → 두 번 등록 방지
     *  - 값 채우기가 끝나지 않았으면 안 됨
     *  - 다른 작업이 어느 전산에서든 저장을 풀어 둔 채 끝나지 않았으면 안 됨(한 번에 한 대)
     */
    canSubmit(jobId, site) {
      const { jobs } = load();
      const j = jobs.get(jobId);
      if (!j) return { ok: false, reason: '작업을 찾을 수 없습니다.' };
      if (j.closed) return { ok: false, reason: '이미 닫힌 작업입니다.' };
      const st = siteState(j, site);
      if (st === 'uncertain') return { ok: false, reason: '저장했는지 확인되지 않았습니다. 목록에서 먼저 확인해 주세요.', uncertain: true };
      if (st === 'stopped') return { ok: false, reason: '멈춘 작업입니다. 정리한 뒤 새 작업으로 다시 해 주세요.' };
      if (st !== 'filled') return { ok: false, reason: st === 'none' ? '아직 값을 채우지 않았습니다.' : '이미 저장 단계를 지났습니다(' + st + ').' };
      //  한 번에 한 대: 저장을 한 번이라도 풀어 준 다른 작업이 닫히기(세 전산 모두 끝) 전에는 막는다.
      //  입력만 해 둔 차(저장을 안 푼 작업)는 기다리는 차라 막지 않는다.
      for (const o of jobs.values()) {
        if (o.jobId === jobId || o.closed || o.office !== j.office) continue;
        const started = Object.keys(o.sites).some((s) => {
          const x = siteState(o, s);
          return x === 'uncertain' || x === 'stopped' || rank(x) >= rank('submitReady');
        });
        if (started) {
          return { ok: false, reason: '앞 작업(' + o.jobId + ' · ' + o.carNo + ')이 아직 끝나지 않았습니다. 한 번에 한 대만 등록합니다.' };
        }
      }
      return { ok: true };
    },

    /** 저장 단추를 풀기 직전에 부른다. 기록이 디스크에 남으면 true. 실패하면 단추를 풀지 말 것. */
    beginSubmit(jobId, site) {
      const c = this.canSubmit(jobId, site);
      if (!c.ok) return c;
      try {
        write({ type: 'stage', jobId, site, stage: 'submitReady' });
      } catch (e) {
        return { ok: false, reason: '기록을 남기지 못해 저장 단추를 풀지 않았습니다: ' + e.message };
      }
      return { ok: true };
    },

    /** 응답이 끊겨 결과를 모를 때 — 상태는 그대로(불확실) 두고 까닭만 남긴다 */
    noteUncertain(jobId, site, reason) {
      write({ type: 'stage', jobId, site, stage: 'submitReady', reason: String(reason || '응답 없음').slice(0, 300) });
    },

    /** 불확실을 풀기: 목록을 확인한 결과(saved / notSaved) */
    resolve(jobId, site, outcome, extra) {
      if (outcome !== 'saved' && outcome !== 'notSaved') throw new Error('확인 결과는 saved 또는 notSaved 여야 합니다.');
      const { jobs } = load();
      const j = jobs.get(jobId);
      if (!j) throw new Error('작업을 찾을 수 없습니다: ' + jobId);
      if (siteState(j, site) !== 'uncertain') throw new Error(site + ' 은(는) 불확실 상태가 아닙니다.');
      write({ type: 'resolve', jobId, site, outcome, ...pick(extra) });
    },

    close(jobId, reason) {
      write({ type: 'close', jobId, ...pick({ reason }) });
    },

    /** 다시 켰을 때 보여 줄 것: 끝나지 않은 작업과 전산별 상태 */
    resumeList() {
      const { jobs, broken } = load();
      const list = [];
      jobs.forEach((j) => {
        if (j.closed) return;
        const sites = {};
        Object.keys(j.sites).forEach((s) => { sites[s] = { state: siteState(j, s), checkNo: j.sites[s].checkNo || null }; });
        if (j.office !== office) return;
        list.push({ jobId: j.jobId, office: j.office, carNo: j.carNo, vin: j.vin, sites });
      });
      return { list, broken };
    },

    get(jobId) {
      const j = load().jobs.get(jobId);
      if (!j) return null;
      const sites = {};
      Object.keys(j.sites).forEach((s) => { sites[s] = { state: siteState(j, s), checkNo: j.sites[s].checkNo || null }; });
      return { jobId: j.jobId, office: j.office, carNo: j.carNo, vin: j.vin, closed: j.closed, sites };
    },
  };
}

module.exports = { createJobLog, STAGES };

/**
 * 점검번호 확인 (설계서 2판 2장) — 전산 화면을 모르는 순수 판단만
 * ------------------------------------------------------------------
 *  원칙
 *   1. 예상(«마지막 번호 + 1»)은 참고 경보일 뿐 안전장치가 아니다.
 *   2. 안전은 실제 번호 확인으로 지킨다.
 *   3. 번호는 «목록 맨 위 줄»이 아니라 우리 차(차량번호 + 차대번호) 줄에서만 읽는다.
 *   4. 같은 차의 예전 점검·재등록 줄은 «저장 전에 찍어 둔 목록»으로 걸러 낸다.
 *   5. 다르거나 읽지 못하면 통과가 아니다(ok:false).
 *
 *  화면에서 읽은 목록 줄 모양(전산 연결 모듈이 만들어 넘김):
 *    { checkNo, carNo, vin, date, status }   — 없는 칸은 비워 둠
 */

const normCar = (s) => String(s || '').replace(/\s+/g, '');
const normVin = (s) => String(s || '').replace(/[\s-]+/g, '').toUpperCase();
const normNo = (s) => String(s || '').replace(/\s+/g, '');

/** 지워진·취소된 줄로 보이는가 (상태 글자로 판단) */
function isDeleted(row) {
  return /삭제|취소|폐기|무효/.test(String((row && row.status) || ''));
}

/**
 * 차대번호가 같은가.
 *  목록에 차대번호 전체(17자리)가 있으면 전체로, 일부만 있으면 뒤 6자리 이상 같을 때만.
 *  목록에 차대번호가 없으면 «모름» — 같다고 치지 않는다.
 */
function vinMatches(rowVin, carVin) {
  const a = normVin(rowVin);
  const b = normVin(carVin);
  if (!a || !b) return null;                      // 모름
  if (a.length >= 17 && b.length >= 17) return a === b;
  const n = Math.min(a.length, b.length);
  if (n < 6) return null;
  return a.slice(-n) === b.slice(-n);
}

/** 이 줄이 우리 차인가: 차량번호 같음 + 차대번호 같음(모르면 아님) */
function isOurCar(row, car) {
  if (!row || normCar(row.carNo) !== normCar(car.carNo)) return false;
  return vinMatches(row.vin, car.vin) === true;
}

/**
 * 저장 전에 찍어 두는 «이 차의 예전 번호들» — 예전 점검·재등록·지운 줄 모두.
 *  차대번호를 모르는 줄도 차량번호가 같으면 넣는다(예전 줄을 새 번호로 잘못 고르지 않게).
 */
function snapshotOldNumbers(rows, car) {
  const set = new Set();
  (rows || []).forEach((r) => {
    if (normCar(r.carNo) === normCar(car.carNo) && normNo(r.checkNo)) set.add(normNo(r.checkNo));
  });
  return set;
}

/**
 * 저장 뒤 목록에서 이번에 새로 생긴 우리 차 번호를 찾는다.
 *  @param rows      저장 뒤 목록 줄
 *  @param car       { carNo, vin }
 *  @param oldNos    snapshotOldNumbers 결과 (저장 전 목록) — 없으면 찾지 않는다
 *  @param since     작업 시작 날짜(YYYY-MM-DD). 그보다 이른 날짜 줄은 예전 것으로 본다
 */
function findNewNumber(rows, car, oldNos, since) {
  if (!(oldNos instanceof Set)) {
    return { ok: false, reason: '저장 전 목록을 찍어 두지 않아 예전 번호와 구분할 수 없습니다.' };
  }
  const sameCar = (rows || []).filter((r) => normCar(r.carNo) === normCar(car.carNo));
  const fresh = sameCar.filter((r) => normNo(r.checkNo) && !oldNos.has(normNo(r.checkNo)) && !isDeleted(r))
    .filter((r) => !(since && r.date && String(r.date).slice(0, 10) < since));
  if (!fresh.length) return { ok: false, reason: '목록에서 이 차의 새 점검번호를 찾지 못했습니다.' };
  if (fresh.length > 1) {
    return { ok: false, reason: '이 차의 새 점검번호로 보이는 줄이 ' + fresh.length + '개입니다(' +
             fresh.map((r) => normNo(r.checkNo)).join(', ') + '). 사람이 확인해 주세요.' };
  }
  const row = fresh[0];
  const v = vinMatches(row.vin, car.vin);
  if (v === null) return { ok: false, reason: '번호 ' + normNo(row.checkNo) + ' 줄의 차대번호를 읽지 못해 이 차인지 확인할 수 없습니다.' };
  if (v === false) return { ok: false, reason: '번호 ' + normNo(row.checkNo) + ' 줄의 차대번호가 이 차와 다릅니다.' };
  return { ok: true, checkNo: normNo(row.checkNo) };
}

/** 두 번호가 같은가 (빈 값이면 통과 아님) */
function sameNumber(label, expected, actual) {
  const a = normNo(expected);
  const b = normNo(actual);
  if (!a) return { ok: false, reason: label + ': 기준 번호(카인포)가 없습니다.' };
  if (!b) return { ok: false, reason: label + ': 번호를 읽지 못했습니다.' };
  if (a !== b) return { ok: false, reason: label + ': 번호가 다릅니다 (기준 ' + a + ' / 읽은 값 ' + b + ').' };
  return { ok: true };
}

/** B. 카모두 저장 직전 — 번호 칸 = 카인포 번호, «자동 부여» 꺼짐 */
function checkCarmodooBeforeSave({ carinfoNo, fieldValue, autoAssignOn }) {
  if (autoAssignOn !== false) {
    return { ok: false, reason: '카모두 «자동 부여»가 꺼졌는지 확인하지 못했습니다.' };
  }
  return sameNumber('카모두 저장 전', carinfoNo, fieldValue);
}

/** C. 카모두 저장 뒤 — 저장된 번호 = 카인포 번호 */
function checkCarmodooAfterSave({ carinfoNo, savedNo }) {
  return sameNumber('카모두 저장 뒤', carinfoNo, savedNo);
}

/** D. 엔트리 저장 직전 — 번호 · 차량번호 · 차대번호 셋 다 */
function checkEntryBeforeSave({ carinfoNo, entryNo, entryCarNo, entryVin, car }) {
  const n = sameNumber('엔트리 번호', carinfoNo, entryNo);
  if (!n.ok) return n;
  if (!normCar(entryCarNo)) return { ok: false, reason: '엔트리가 끌어온 차량번호를 읽지 못했습니다.' };
  if (normCar(entryCarNo) !== normCar(car.carNo)) {
    return { ok: false, reason: '엔트리가 끌어온 차량번호(' + normCar(entryCarNo) + ')가 이 차(' + normCar(car.carNo) + ')와 다릅니다.' };
  }
  const v = vinMatches(entryVin, car.vin);
  if (v === null) return { ok: false, reason: '엔트리가 끌어온 차대번호를 읽지 못했습니다.' };
  if (v === false) return { ok: false, reason: '엔트리가 끌어온 차대번호가 이 차와 다릅니다.' };
  return { ok: true };
}

/**
 * 참고 경보 — 카인포 저장 전, «카인포 마지막 번호 + 1» 과 «엔트리 다음 번호» 비교.
 *  안전장치가 아니다. 다르면 저장 전에 사람에게 물어보는 용도.
 *  번호 모양을 모르면(숫자로 끝나지 않음 등) 'unknown'.
 */
function previewNext(lastCarinfoNo, entryNextNo) {
  const m = /^(.*?)(\d+)$/.exec(normNo(lastCarinfoNo));
  const e = normNo(entryNextNo);
  if (!m || !e) return { level: 'unknown', reason: '번호 모양을 몰라 미리 비교하지 못했습니다.' };
  const next = m[1] + String(Number(m[2]) + 1).padStart(m[2].length, '0');
  if (next === e) return { level: 'same', expected: next };
  return { level: 'warn', expected: next, entry: e,
           reason: '예상 카인포 번호 ' + next + ' 와 엔트리 다음 번호 ' + e + ' 가 다릅니다. 저장하기 전에 확인해 주세요.' };
}

module.exports = {
  normCar, normVin, normNo, vinMatches, isOurCar, isDeleted,
  snapshotOldNumbers, findNewNumber,
  checkCarmodooBeforeSave, checkCarmodooAfterSave, checkEntryBeforeSave,
  previewNext,
};
