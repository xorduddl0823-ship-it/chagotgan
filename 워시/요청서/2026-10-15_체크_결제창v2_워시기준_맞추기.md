[작업 요청서]
보내는 곳: 워시 방
받는 곳:   체크 방 (총괄 방 거쳐)
제목:      결제창 v2 — 워시 v120 기준 코드로 맞추기 (규칙 3-2)

1. 왜 필요한가요?
   체크 v30(fb4b33c)과 워시 v120(4b2cb5d)의 결제창 v2 를 견줘 보니 거의 같고, 두 군데가 달라요. 워시가 기준이라 워시 쪽으로 맞춰 주세요.
   ① 비회원 결제 customerKey — 서버(sb10)는 글자 "ANONYMOUS" 를 줘요. 토스 v2 안내는 **SDK 상수 `TossPayments.ANONYMOUS`** 를 넘기라고 해요.
      체크 v30 은 글자 "ANONYMOUS" 를 그대로 넘겨서, 토스가 «ANONYMOUS 라는 이름의 회원»으로 볼 수 있어요. 워시는 상수로 바꿔 넘겨요.
   ② loadToss — 한 화면에서 v1 을 이미 불러온 뒤 v2 를 또 부르면 두 SDK 가 겹쳐요. 워시는 «새로고침해 주세요»로 멈춰요.
      (자동 결제 카드 등록창도 이 loadToss("v2") 를 같이 써요)
2. 무엇을 해 주면 되나요?
   체크의 `loadToss` 부터 `cwPay` 끝까지를 아래 코드로 바꿔 주세요 (워시 v120 14832~14892줄 그대로). `afterPay` 는 두 제품이 이미 같아요.
   - `window.TossPayments.__preview` 는 미리보기 가짜 토스용 표시예요. 진짜 화면에서는 아무 일도 안 해요.
3. 끝났다고 보는 기준  두 파일의 이 부분이 한 글자도 다르지 않음
4. 기다리는 다른 일  토스 테스트 모드 시험 (대표님 테스트 열쇠)
5. 보안 검토 필요?  예 (작음 · 결제창)

```js
  /* v120: 서버(sb10)가 sdk:"v2" 를 주면 v2 창, 옛 서버면 v1 창 */
  var TOSSVER = "";
  function loadToss(ver){
    ver = ver === "v2" ? "v2" : "v1";
    return new Promise(function(res, rej){
      if(window.TossPayments && (TOSSVER === ver || window.TossPayments.__preview)) return res(window.TossPayments);
      if(TOSSVER && TOSSVER !== ver) return rej({message:"결제 창을 다시 열려면 화면을 새로고침해 주세요"});
      TOSSVER = ver;
      var sc = document.createElement("script");
      sc.src = ver === "v2" ? "https://js.tosspayments.com/v2/standard" : "https://js.tosspayments.com/v1/payment";
      sc.onload = function(){ res(window.TossPayments); };
      sc.onerror = function(){ rej({message:"카드 결제 창을 불러오지 못했습니다"}); };
      document.head.appendChild(sc);
    });
  }
  var MSGID = "cwMsg";
  function cwMsg(html){
    var el = document.getElementById(MSGID) || document.getElementById("cwMsg");
    if(el) el.innerHTML = html;
  }
  function cwPay(kind, amount, useCash, msgId){
    MSGID = msgId || "cwMsg";
    cwMsg('<span style="color:#667389">결제 준비 중입니다…</span>');
    cwRpc("payStart",[kind, amount, useCash]).then(function(r){
      if(!r || r.ok === false){
        cwMsg('<span style="color:#B45309">'+esc2((r&&r.note)||"카드 결제는 아직 준비 중입니다")+'</span>');
        return;
      }
      if(r.done){   /* 캐시만으로 다 낸 경우 */
        cwMsg('<span style="color:#1A7F4B">캐시로 결제했습니다.</span>');
        try{ window.ccwCashRefresh(); }catch(e){}
        return;
      }
      try{ localStorage.setItem("ccw_pay", JSON.stringify({orderId:r.orderId, amount:r.amount, kind:kind})); }catch(e){}
      var v2 = r.sdk === "v2";
      return loadToss(v2 ? "v2" : "v1").then(function(TP){
        var back = location.origin + location.pathname;
        var tp = TP(r.clientKey);
        if(v2){
          var ck = (!r.customerKey || r.customerKey === "ANONYMOUS") ? TP.ANONYMOUS : r.customerKey;
          return tp.payment({customerKey: ck}).requestPayment({
            method: "CARD",
            amount: {currency: "KRW", value: r.amount},
            orderId: r.orderId, orderName: r.orderName,
            customerName: r.customerName,
            successUrl: back + "?pay=ok",
            failUrl:    back + "?pay=no"
          });
        }
        return tp.requestPayment("카드", {
          amount: r.amount, orderId: r.orderId, orderName: r.orderName,
          customerName: r.customerName,
          successUrl: back + "?pay=ok",
          failUrl:    back + "?pay=no"
        });
      });
    }).catch(function(e){
      cwMsg('<span style="color:#C0392B">'+esc2((e&&e.message)||"결제하지 못했습니다")+'</span>');
    });
  }
```
